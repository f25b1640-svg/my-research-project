                    <p>
                        Malaysia currently operates at approximately <strong>65% to 70% Self-Sufficiency Level (SSL)</strong> for rice, relying on imports from neighboring countries to close the supply gap. Under the <strong class="text-slate-900">National Agrofood Policy 2.0 (DPN 2.0)</strong>, the Ministry of Agriculture and Food Security has prioritized boosting crop productivity using Smart Farming technologies. 
                    </p>
                    <p>
                        To achieve high yields, farmers must manage the growth cycle of flagship paddy varieties like <strong>MR219</strong> (developed by MARDI). Paddy growth typically follows a sigmoidal (S-shaped) logistic curve, consisting of active tillering (vegetative phase), panicle development (booting phase), flowering, and final grain maturity. 
                    </p>
                    <p class="font-semibold text-emerald-800 bg-emerald-50 p-3 rounded-lg border-l-4 border-emerald-500">
                        The crucial bioscience question is: <strong>When does the plant accumulate biomass at its absolute maximum rate?</strong> This peak moment represents the inflection point on the logistic growth curve, where the derivative \(f'(x)\) is maximized. Timing high-nitrogen top-dressing (e.g., Urea) precisely at or slightly before this peak rate ensures maximum nutrient assimilation, optimal stem thickness, and significantly enhanced panicle count.
                    </p>

                            </svg>

                        Biomathematical Model
                    </h3>
                    
                    <div class="space-y-4 text-sm text-slate-600">
                        <div>
                            <span class="block font-semibold text-xs text-slate-500 uppercase tracking-wide">1. Logistic Growth Function: \(f(x)\)</span>
                            <div class="bg-indigo-50/50 p-3 rounded-lg my-2 font-mono text-center text-indigo-900 overflow-x-auto text-base">
                                $$f(x) = \frac{L}{1 + e^{-k(x - x_0)}}$$
                            </div>
                            <p class="text-[11px] text-slate-500">
                                Models overall paddy height (cm) at day \(x\). \(L\) is the max capacity, \(k\) is the steepness (growth coefficient), and \(x_0\) is the midpoint of rapid growth.
                            </p>
                        </div>

                        <div>
                            <span class="block font-semibold text-xs text-slate-500 uppercase tracking-wide">2. Growth Rate Derivative: \(f'(x)\)</span>
                            <div class="bg-emerald-50/50 p-3 rounded-lg my-2 font-mono text-center text-emerald-900 overflow-x-auto text-base">
                                $$f'(x) = \frac{L \cdot k \cdot e^{-k(x - x_0)}}{(1 + e^{-k(x - x_0)})^2}$$
                            </div>
                            <p class="text-[11px] text-slate-500">
                                Solved via chain and quotient rules. It calculates the instantaneous rate of elongation (cm per day) at exactly Day \(x\).
                            </p>
                        </div>
                    </div>
                </div>

                <div class="mt-6 pt-4 border-t border-slate-100 bg-amber-50/60 p-3 rounded-xl border border-amber-100">
                    <span class="block font-bold text-amber-800 text-xs uppercase tracking-wider mb-1">Calculus Fact Sheet</span>
                    <p class="text-xs text-amber-900 leading-relaxed">
                        At \(x = x_0\), the rate of change \(f'(x)\) is maximized. Mathematically, this is where the second derivative \(f''(x) = 0\), identifying the biological <strong>point of inflection</strong>.
                    </p>
                </div>
            </div>

        // Global variables for model parameters
        let paramL = 110;  // Max height (cm)
        let paramK = 0.08; // Growth rate steepness
        let paramX0 = 55;  // Inflexion day point
        
        let chartInstance = null;
        let animationTimer = null;
        let currentAnimationDay = 0;

        // Mathematical Equations
        // Logistic growth height f(x)
        function f(x, L, k, x0) {
            return L / (1 + Math.exp(-k * (x - x0)));
        }

        // Instantaneous growth rate f'(x)
        function f_prime(x, L, k, x0) {
            const expTerm = Math.exp(-k * (x - x0));
            return (L * k * expTerm) / Math.pow(1 + expTerm, 2);
        }

        // Create full datasets for range Day 0 to 115
        function generateData(L, k, x0) {
            const labels = [];
            const heightData = [];
            const rateData = [];

            for (let day = 0; day <= 115; day++) {
                labels.push(day);
                heightData.push(parseFloat(f(day, L, k, x0).toFixed(2)));
                rateData.push(parseFloat(f_prime(day, L, k, x0).toFixed(3)));
            }

            return { labels, heightData, rateData };
        }

        // Function to set slider presets
        function setPreset(L, k, x0) {
            document.getElementById('heightL').value = L;
            document.getElementById('growthK').value = k;
            document.getElementById('midX0').value = x0;

            updateParamsFromSliders();
        }

        // Function to update state and re-render everything
        function updateParamsFromSliders() {
            paramL = parseFloat(document.getElementById('heightL').value);
            paramK = parseFloat(document.getElementById('growthK').value);
            paramX0 = parseFloat(document.getElementById('midX0').value);

            // Update visible label numbers
            document.getElementById('valL').textContent = paramL;
            document.getElementById('valK').textContent = paramK;
            document.getElementById('valX0').textContent = paramX0;

            updateChart();
        }

        // Dynamic evaluation calculation and agronomical interpretation generator
        function evaluateDayMetric(day) {
            const height = f(day, paramL, paramK, paramX0);
            const rate = f_prime(day, paramL, paramK, paramX0);

            // Calculate inflection rate peak check
            const peakRateVal = (paramL * paramK) / 4; // Peak of derivative is L*k/4 at x0

            // Update UI Readout Elements
            document.getElementById('readoutDay').textContent = `Day ${day}`;
            document.getElementById('readoutHeight').textContent = height.toFixed(1);
            document.getElementById('readoutRate').textContent = rate.toFixed(2);

            // Update Highlight point in Chart dataset
            if (chartInstance) {
                chartInstance.data.datasets[2].data = [{ x: day, y: parseFloat(height.toFixed(1)) }];
                chartInstance.update('none');
            }

            // Phase definitions & Layman agronomy diagnostics
            let phaseName = "";
            let advice = "";

            if (day >= 0 && day <= 20) {
                phaseName = "Vegetative: Seedling & Rooting";
                advice = `At Day ${day}, the MR219 seedling is recovering from transplanting shock. The growth rate f'(x) is low (${rate.toFixed(2)} cm/day) as energy is directed downward to establish extensive rooting networks in the Kedah clay-soil. Keep irrigation minimal (2-3 cm depth) to encourage root anchoring. Do not apply heavy inputs yet.`;
            } else if (day > 20 && day <= 45) {
                phaseName = "Vegetative: Tillering Phase";
                advice = `At Day ${day}, the plant height is rising as the growth curve gains momentum. The derivative rate is accelerating up to ${(rate).toFixed(2)} cm/day. The paddy is branching (producing active tillers). This is the optimal window for the 1st Nitrogen split application (e.g. Urea compound) to yield healthy stem numbers.`;
            } else if (day > 45 && day <= 65) {
                phaseName = "Reproductive: Booting & Heading";
                const isInflection = Math.abs(day - paramX0) <= 2;
                if (isInflection) {
                    phaseName = "Reproductive: Inflection Point! (Panicle Peak)";
                    advice = `CRITICAL BIOLOGICAL MILESTONE! At Day ${day}, we hit the exact INFLECTION POINT of our Sigmoid Growth Curve. Paddy elongation rate f'(x) reaches its ABSOLUTE MAXIMUM at ${rate.toFixed(2)} cm/day. Cellular expansion of the growing panicle flower within the stalk is explosive. Top-dress with high Potassium/NPK immediately to maximize final grain length and crop volume!`;
                } else {
                    advice = `At Day ${day}, the crop is in rapid stem elongation and reproductive preparation. The derivative rate is exceptionally high at ${rate.toFixed(2)} cm/day. Ensure adequate standing water (5-10 cm) to protect developing internal panicles against heat stress.`;
                }
            } else if (day > 65 && day <= 85) {
                phaseName = "Reproductive: Flowering & Milk Phase";
                advice = `At Day ${day}, the plant height is stabilizing around ${(height).toFixed(1)} cm. Notice how the derivative f'(x) drops to ${rate.toFixed(2)} cm/day. Stalk growth slows because photosynthetic assets are redirected entirely into filling the emerging grains with milky starch. Prevent water drainage during this vital fertilization window.`;
            } else {
                phaseName = "Maturation: Grain Ripening & Harvest";
                advice = `At Day ${day}, growth is complete. The crop derivative f'(x) is near zero (${rate.toFixed(2)} cm/day), showing static height. The grains are turning golden-bronze. Drain water completely from the rice field plots 10-14 days before harvest day to facilitate soil drying, allowing heavy rice combine harvesters to traverse the Sekinchan estates safely.`;
            }

            document.getElementById('readoutPhase').textContent = phaseName;
            document.getElementById('readoutAdvice').innerHTML = advice;
            document.getElementById('readoutTitle').textContent = `Day ${day} Diagnosis`;
        }

        // Setup play simulation feature
        function playTimelapse() {
            if (animationTimer) {
                // Pause if running
                clearInterval(animationTimer);
                animationTimer = null;
                document.getElementById('playText').textContent = "Resume Timelapse";
                document.getElementById('playIcon').innerHTML = `<svg class="w-3.5 h-3.5" fill="currentColor" viewBox="0 0 20 20" xmlns="http://www.w3.org/2000/svg"><path fill-rule="evenodd" d="M10 18a8 8 0 100-16 8 8 0 000 16zM9.555 7.168A1 1 0 008 8v4a1 1 0 001.555.832l3-2a1 1 0 000-1.664l-3-2z" clip-rule="evenodd"></path></svg>`;
            } else {
                // Reset to day 0 if starting at the end
                if (currentAnimationDay >= 115) {
                    currentAnimationDay = 0;
                }
                document.getElementById('playText').textContent = "Pause Timelapse";
                document.getElementById('playIcon').innerHTML = `<svg class="w-3.5 h-3.5" fill="currentColor" viewBox="0 0 20 20" xmlns="http://www.w3.org/2000/svg"><path fill-rule="evenodd" d="M18 10a8 8 0 11-16 0 8 8 0 0116 0zM7 8a1 1 0 012 0v4a1 1 0 11-2 0V8zm5-1a1 1 0 00-1 1v4a1 1 0 102 0V8a1 1 0 00-1-1z" clip-rule="evenodd"></path></svg>`;

                animationTimer = setInterval(() => {
                    evaluateDayMetric(currentAnimationDay);
                    currentAnimationDay++;
                    if (currentAnimationDay > 115) {
                        clearInterval(animationTimer);
                        animationTimer = null;
                        document.getElementById('playText').textContent = "Restart Timelapse";
                        document.getElementById('playIcon').innerHTML = `<svg class="w-3.5 h-3.5" fill="currentColor" viewBox="0 0 20 20" xmlns="http://www.w3.org/2000/svg"><path fill-rule="evenodd" d="M4 4v5h.582m15.356 2A8.001 8.001 0 1121.21 3.562M9 11h11m0 0v11m0-11l-10 10"></path></svg>`;
                    }
                }, 100);
            }
        }
